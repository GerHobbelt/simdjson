274 // ============================================================
275 // Internal hooks for the position-as-local writer in json_builder.h.
276 // These exist so the reflection atom code can hold buffer pointer,
277 // position and capacity in registers across long write chains rather
278 // than reloading them after every char* write (strict aliasing
279 // forces those reloads when accessed via members of *this). User
280 // code should NOT call these directly.
281 // ============================================================
282 simdjson_inline char *unsafe_data() noexcept { return buffer_data(); }

284 simdjson_inline size_t unsafe_capacity() const noexcept { return capacity; }
285 simdjson_inline void unsafe_set_position(size_t p) noexcept { position = p; }
288 simdjson_inline bool unsafe_grow(size_t needed_total_capacity) noexcept {
289 grow_buffer(needed_total_capacity);
