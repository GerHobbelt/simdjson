106inline simdjson_result<element> object::at_pointer(std::string_view json_pointer) const noexcept {
107 SIMDJSON_DEVELOPMENT_ASSERT(tape.usable()); // https://github.com/simdjson/simdjson/issues/1914
108 if(json_pointer.empty()) { // an empty string means that we return the current node
109 return element(this->tape); // copy the current node
110 } else if(json_pointer[0] != '/') { // otherwise there is an error
111 return INVALID_JSON_POINTER;
112 }
113 json_pointer = json_pointer.substr(1);
114 size_t slash = json_pointer.find('/');
115 std::string_view key = json_pointer.substr(0, slash);
116 // Grab the child with the given key
117 simdjson_result<element> child;
118
119 // If there is an escape character in the key, unescape it and then get the child.
120 size_t escape = key.find('~');
121 if (escape != std::string_view::npos) {
122 // Unescape the key
123 std::string unescaped(key);
124 do {
125 switch (unescaped[escape+1]) {
126 case '0':
127 unescaped.replace(escape, 2, "~");
128 break;
129 case '1':
130 unescaped.replace(escape, 2, "/");
131 break;
132 default:
133 return INVALID_JSON_POINTER; // "Unexpected ~ escape character in JSON pointer");
134 }
135 escape = unescaped.find('~', escape+1);
136 } while (escape != std::string::npos);
137 child = at_key(unescaped);
138 } else {
139 child = at_key(key);
140 }
141 if(child.error()) {
142 return child; // we do not continue if there was an error
143 }
144 // If there is a /, we have to recurse and look up more of the path
145 if (slash != std::string_view::npos) {
146 child = child.at_pointer(json_pointer.substr(slash));
147 }
148 return child;
149}

157inline void object::process_json_path_of_child_elements(std::vector<element>::iterator& current, std::vector<element>::iterator& end, const std::string_view& path_suffix, std::vector<element>& accumulator) const noexcept {
158 if (current == end) {
159 return;
160 }
161
162 simdjson_result<std::vector<element>> result;
163
164 for (auto it = current; it != end; ++it) {
165 std::vector<element> child_result;
166 auto error = it->at_path_with_wildcard(path_suffix).get(child_result);
167 if(error) {
168 continue;
169 }
170 accumulator.insert(accumulator.end(),
171 std::make_move_iterator(child_result.begin()),
172 std::make_move_iterator(child_result.end()));
173 }
174}

181inline simdjson_result<element> object::at_key(std::string_view key) const noexcept {
182 iterator end_field = end();
183 for (iterator field = begin(); field != end_field; ++field) {
184 if (field.key_equals(key)) {
185 return field.value();
186 }
187 }
188 return NO_SUCH_FIELD;

191inline std::vector<element>& object::get_values(std::vector<element>& out) const noexcept {
192 iterator end_field = end();
193 iterator begin_field = begin();
194
195 out.reserve(std::distance(begin_field, end_field));
196 for (iterator field = begin_field; field != end_field; ++field) {
197 out.emplace_back(field.value());
198 }
199
200 return out;
201}

280inline bool object::iterator::key_equals(std::string_view o) const noexcept {
281 // We use the fact that the key length can be computed quickly
282 // without access to the string buffer.
283 const uint32_t len = key_length();
284 if(o.size() == len) {
285 // We avoid construction of a temporary string_view instance.
286 return (memcmp(o.data(), key_c_str(), len) == 0);
287 }
288 return false;
289}

291inline bool object::iterator::key_equals_case_insensitive(std::string_view o) const noexcept {
292 // We use the fact that the key length can be computed quickly
293 // without access to the string buffer.
294 const uint32_t len = key_length();
295 if(o.size() == len) {
296 // See For case-insensitive string comparisons, avoid char-by-char functions
297 // https://lemire.me/blog/2020/04/30/for-case-insensitive-string-comparisons-avoid-char-by-char-functions/
298 // Note that it might be worth rolling our own strncasecmp function, with vectorization.
299 return (simdjson_strncasecmp(o.data(), key_c_str(), len) == 0);
300 }
301 return false;
302}
