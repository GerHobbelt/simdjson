21simdjson_inline error_code parse_json_pointer_array_index(std::string_view json_pointer,
22 size_t &array_index,
23 size_t &token_length) noexcept {
24 array_index = 0;
25 token_length = 0;
26
27 for (; token_length < json_pointer.length() && json_pointer[token_length] != '/';
28 token_length++) {
29 uint8_t digit = uint8_t(json_pointer[token_length] - '0');
30 // Check for non-digit in array index. If it's there, we're trying to get a field in an object.
31 if (digit > 9) {
32 return INCORRECT_TYPE;
33 }
34 // 0 followed by other digits is invalid.
35 if (token_length > 0 && json_pointer[0] == '0') {
36 return INVALID_JSON_POINTER;
37 }
38 if (array_index >
39 (((std::numeric_limits<size_t>::max)() - digit) / 10)) {
40 return INDEX_OUT_OF_BOUNDS;
41 }
42 array_index = array_index * 10 + digit;
43 }
44
45 // Empty string is invalid; so is a "/" with no digits before it.
46 if (token_length == 0) {
47 return INVALID_JSON_POINTER;
48 }
49
50 return SUCCESS;
51}

71inline std::string json_path_to_pointer_conversion(std::string_view json_path) {
72 size_t i = 0;
73 // if JSONPath starts with $, skip it
74 // json_path.starts_with('$') requires C++20.
75 if (!json_path.empty() && json_path.front() == '$') {
76 i = 1;
77 }
78 if (i >= json_path.size() || (json_path[i] != '.' &&
79 json_path[i] != '[')) {
80 return "-1"; // This is just a sentinel value, the caller should check for this and return an error.
81 }
82
83 std::string result;
84 // Reserve space to reduce allocations, adjusting for potential increases due
85 // to escaping.
86 result.reserve(json_path.size() * 2);
87
88 while (i < json_path.length()) {
89 if (json_path[i] == '.') {
90 result += '/';
91 } else if (json_path[i] == '[') {
92 result += '/';
93 ++i; // Move past the '['
94 while (i < json_path.length() && json_path[i] != ']') {
95 if (json_path[i] == '~') {
96 result += "~0";
97 } else if (json_path[i] == '/') {
98 result += "~1";
99 } else {
100 result += json_path[i];
101 }
102 ++i;
103 }
104 if (i == json_path.length() || json_path[i] != ']') {
105 return "-1"; // Using sentinel value that will be handled as an error by the caller.
106 }
107 } else {
108 if (json_path[i] == '~') {
109 result += "~0";
110 } else if (json_path[i] == '/') {
111 result += "~1";
112 } else {
113 result += json_path[i];
114 }
115 }
116 ++i;
117 }
118
119 return result;
120}

122inline std::pair<std::string_view, std::string_view> get_next_key_and_json_path(std::string_view& json_path) {
123 std::string_view key;
124
125 if (json_path.empty()) {
126 return {key, json_path};
127 }
128 size_t i = 0;
129
130 // if JSONPath starts with $, skip it
131 if (json_path.front() == '$') {
132 i = 1;
133 }
134
135
136 if (i < json_path.length() && json_path[i] == '.') {
137 i += 1;
138 size_t key_start = i;
139
140 while (i < json_path.length() && json_path[i] != '[' && json_path[i] != '.') {
141 ++i;
142 }
143
144 key = json_path.substr(key_start, i - key_start);
145 } else if ((i + 1 < json_path.size()) && json_path[i] == '[' &&
146 (json_path[i + 1] == '\'' || json_path[i + 1] == '"')) {
147 // Bracket-quoted key: ['key'] or ["key"].
148 // Require a matching closing quote and a following ']'. If either is
149 // missing, return an empty key and the original path so callers can treat
150 // this as a parse failure (e.g. INVALID_JSON_POINTER) without advancing.
151 // Without this check, i += 2 can make i > size() and substr throws
152 // std::out_of_range, which aborts noexcept callers such as
153 // at_path_with_wildcard / for_each_at_path_with_wildcard.
154 const char quote = json_path[i + 1];
155 i += 2;
156 const size_t key_start = i;
157 while (i < json_path.length() && json_path[i] != quote) {
158 ++i;
159 }
160 if (i >= json_path.length() || // missing closing quote
161 i + 1 >= json_path.length() || // missing ]
162 json_path[i + 1] != ']') {
163 return {key, json_path};
164 }
165 key = json_path.substr(key_start, i - key_start);
166 i += 2; // past quote and ]
167 } else if ((i+2 < json_path.size()) && json_path[i] == '[' && json_path[i+1] == '*' && json_path[i+2] == ']') { // i.e [*].additional_keys or [*]["additional_keys"]
168 key = "*";
169 i += 3;
170 } else if (i < json_path.size() && json_path[i] == '[') { // i.e. [0] or [123]
171 size_t j = i + 1;
172 while (j < json_path.size() && json_path[j] >= '0' && json_path[j] <= '9') {
173 ++j;
174 }
175 if (j > i + 1 && j < json_path.size() && json_path[j] == ']') {
176 key = json_path.substr(i + 1, j - i - 1);
177 i = j + 1;
178 }
179 }
180
181
182 return std::make_pair(key, json_path.substr(i));
183}

190struct json_path_segment {
191 // The key or array index at this level, "*" for a wildcard. Empty if the
192 // path could not be parsed at this level.
193 std::string_view key{};
194 // Whether a '*' occurs in the rest of the path, after this segment. A
195 // segment that matches nothing before a wildcard yields an empty result
196 // instead of an error, as in DOM's at_path_with_wildcard. Like DOM, this
197 // treats a '*' inside a quoted key as a wildcard.
198 bool wildcard_follows{false};
199};
200
214inline std::vector<json_path_segment> split_json_path(std::string_view json_path) {
215 std::vector<json_path_segment> segments;
216 // The remaining path (always a suffix of json_path) contains a '*' if and
217 // only if it starts at or before the last one.
218 const size_t last_star = json_path.rfind('*');
219 std::string_view path = json_path;
220 do {
221 std::pair<std::string_view, std::string_view> key_and_json_path = get_next_key_and_json_path(path);
222 path = key_and_json_path.second;
223 const size_t offset = json_path.size() - path.size();
224 json_path_segment segment;
225 segment.key = key_and_json_path.first;
226 segment.wildcard_follows = last_star != std::string_view::npos && last_star >= offset;
227 segments.push_back(segment);
228 if (segment.key.empty()) { break; }
229 } while (!path.empty());
230 return segments;
231}

242struct wildcard_json_path {
243 enum class tail_type { invalid, all_values, pointer };
244 // One JSON pointer ("/key") per leading segment, "/*" being a wildcard.
245 std::vector<std::string> steps{};
246 // What to do once the steps are exhausted.
247 tail_type tail{tail_type::invalid};
248 // The rest of the path, as a JSON pointer, when tail == pointer.
249 std::string tail_pointer{};
250};
251
256inline wildcard_json_path parse_wildcard_json_path(std::string_view json_path) {
257 wildcard_json_path result;
258 // Each key is consumed by at_pointer, which unescapes ~0/~1, so '~' and
259 // '/' in the key must be escaped as json_path_to_pointer_conversion does:
260 // a raw '/' splits the step into two pointer tokens and a raw '~'
261 // invalidates it, either way addressing the wrong element.
262 const auto append_key = [](std::string &pointer, std::string_view key) {
263 pointer += '/';
264 for (const char c : key) {
265 if (c == '~') {
266 pointer += "~0";
267 } else if (c == '/') {
268 pointer += "~1";
269 } else {
270 pointer += c;
271 }
272 }
273 };
274 // The remaining path (always a suffix of json_path) contains a '*' if and
275 // only if it starts at or before the last one.
276 const size_t last_star = json_path.rfind('*');
277 std::string_view path = json_path;
278 while (true) {
279 size_t i = 0;
280 if (!path.empty() && path.front() == '$') {
281 i = 1;
282 }
283 if (i >= path.size() || (path[i] != '.' && path[i] != '[')) {
284 result.tail = wildcard_json_path::tail_type::invalid;
285 return result;
286 }
287 const size_t offset = json_path.size() - path.size();
288 if (last_star == std::string_view::npos || last_star < offset) {
289 // Split the rest like the steps, so that a bracket-quoted key (['b'],
290 // ["b"]) is the key b, not 'b' with its quotes. What we do not split
291 // (e.g. [b], the key "b") is converted as at_path does.
292 std::string_view rest = path;
293 while (!rest.empty()) {
294 std::pair<std::string_view, std::string_view> key_and_json_path = get_next_key_and_json_path(rest);
295 if (key_and_json_path.first.empty()) {
296 const std::string pointer = json_path_to_pointer_conversion(rest);
297 if (pointer == "-1") {
298 result.tail_pointer = pointer;
299 } else {
300 result.tail_pointer += pointer;
301 }
302 break;
303 }
304 append_key(result.tail_pointer, key_and_json_path.first);
305 rest = key_and_json_path.second;
306 }
307 result.tail = (result.tail_pointer == "-1")
308 ? wildcard_json_path::tail_type::invalid
309 : wildcard_json_path::tail_type::pointer;
310 return result;
311 }
312 if ((path.compare(i, 3, "[*]") == 0 && path.size() == i + 3) ||
313 (path.compare(i, 2, ".*") == 0 && path.size() == i + 2)) {
314 result.tail = wildcard_json_path::tail_type::all_values;
315 return result;
316 }
317 std::pair<std::string_view, std::string_view> key_and_json_path = get_next_key_and_json_path(path);
318 if (key_and_json_path.first.empty()) {
319 result.tail = wildcard_json_path::tail_type::invalid;
320 return result;
321 }
322 std::string step;
323 step.reserve(key_and_json_path.first.size() + 1);
324 append_key(step, key_and_json_path.first);
325 result.steps.push_back(std::move(step));
326 path = key_and_json_path.second;
327 // The segment above consumed the path. An empty remainder is the node
328 // that segment selected (at_pointer("")), not a parse error. The check
329 // at the top of the loop is for a non-empty remainder that does not
330 // continue with '.' or '['. A '*' inside the final segment ($.a.b*,
331 // $['b*']) belongs to that segment.
332 if (path.empty()) {
333 result.tail = wildcard_json_path::tail_type::pointer;
334 return result;
335 }
336 }
337}
